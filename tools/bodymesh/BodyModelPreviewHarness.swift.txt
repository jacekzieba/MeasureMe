    // MARK: - Rendering

    /// Renders one body at the given yaw. Reuses one hosting controller per
    /// profile so the 13 380-vertex deform happens once, not once per angle.
    private func render(profile: Profile, yaws: [Double]) async throws -> [UIImage] {
        let parameters = BodyMeshSolver.solve(snapshot: profile.snapshot)

        let view = ZStack {
            Color(white: 0.97)
            MannequinView(parameters: parameters, gender: profile.gender)
        }
        .frame(width: Self.cell.width, height: Self.cell.height)

        let vc = UIHostingController(rootView: view)
        vc.overrideUserInterfaceStyle = .light
        vc.view.frame = CGRect(origin: .zero, size: Self.cell)
        let window = UIWindow(frame: vc.view.frame)
        window.rootViewController = vc
        window.makeKeyAndVisible()
        vc.view.setNeedsLayout()
        vc.view.layoutIfNeeded()
        try await Task.sleep(for: .milliseconds(900))

        guard let scnView = Self.findSCNView(in: vc.view) else {
            XCTFail("No SCNView for \(profile.name)")
            return []
        }
        guard let bodyNode = scnView.scene?.rootNode.childNode(withName: "body", recursively: false),
              bodyNode.geometry != nil else {
            XCTFail("No geometry for \(profile.name) — base mesh not prepared?")
            return []
        }
        scnView.backgroundColor = UIColor(white: 0.97, alpha: 1)
        scnView.scene?.background.contents = UIColor(white: 0.97, alpha: 1)

        var images: [UIImage] = []
        for yaw in yaws {
            bodyNode.eulerAngles.y = Float(yaw)
            try await Task.sleep(for: .milliseconds(120))
            images.append(scnView.snapshot())
        }
        window.isHidden = true
        return images
    }

    private static func findSCNView(in view: UIView) -> SCNView? {
        if let scn = view as? SCNView { return scn }
        for sub in view.subviews {
            if let found = findSCNView(in: sub) { return found }
        }
        return nil
    }

    /// Columns = profiles, rows = viewing angles, plus a caption strip on top.
    private func contactSheet(title: String, columns: [(String, [UIImage])]) -> UIImage {
        let header: CGFloat = 56
        let caption: CGFloat = 44
        let rows = columns.first?.1.count ?? 1
        let size = CGSize(
            width: Self.cell.width * CGFloat(columns.count),
            height: header + caption + Self.cell.height * CGFloat(rows)
        )
        return UIGraphicsImageRenderer(size: size).image { ctx in
            UIColor.white.setFill()
            ctx.fill(CGRect(origin: .zero, size: size))

            (title as NSString).draw(
                at: CGPoint(x: 20, y: 16),
                withAttributes: [
                    .font: UIFont.systemFont(ofSize: 26, weight: .bold),
                    .foregroundColor: UIColor.black
                ]
            )

            let style = NSMutableParagraphStyle()
            style.alignment = .center
            for (index, column) in columns.enumerated() {
                let x = Self.cell.width * CGFloat(index)
                (column.0 as NSString).draw(
                    in: CGRect(x: x, y: header, width: Self.cell.width, height: caption),
                    withAttributes: [
                        .font: UIFont.systemFont(ofSize: 14, weight: .semibold),
                        .foregroundColor: UIColor.darkGray,
                        .paragraphStyle: style
                    ]
                )
                for (row, image) in column.1.enumerated() {
                    image.draw(in: CGRect(
                        x: x,
                        y: header + caption + Self.cell.height * CGFloat(row),
                        width: Self.cell.width,
                        height: Self.cell.height
                    ))
                }
                UIColor(white: 0.85, alpha: 1).setStroke()
                let separator = UIBezierPath()
                separator.move(to: CGPoint(x: x, y: header))
                separator.addLine(to: CGPoint(x: x, y: size.height))
                separator.stroke()
            }
        }
    }

    func testRenderDummyBodyMatrix() async throws {
        await BodyBaseMeshProvider.prepare(for: .male)
        await BodyBaseMeshProvider.prepare(for: .female)

        let directory = outputDirectory()
        let yaws: [Double] = [0, .pi / 4, .pi / 2, 3 * .pi / 4]

        for (title, profiles) in [("Mezczyzna", Self.males), ("Kobieta", Self.females)] {
            var columns: [(String, [UIImage])] = []
            for profile in profiles {
                let images = try await render(profile: profile, yaws: yaws)
                XCTAssertEqual(images.count, yaws.count)
                columns.append((profile.name.replacingOccurrences(of: "\n", with: " — "), images))
            }
            let sheet = contactSheet(title: "\(title) — 0 / 45 / 90 / 135 st.", columns: columns)
            let url = directory.appendingPathComponent(
                title == "Mezczyzna" ? "male-matrix.png" : "female-matrix.png"
            )
            try XCTUnwrap(sheet.pngData()).write(to: url)
            print("📸 WROTE \(url.path)")
        }
    }

    /// Renders the base mesh with NO deformation at all — positions are the
    /// baked ones times stature, normals recomputed the same way the renderer
    /// does. If a crease shows up here it belongs to the asset or the lighting,
    /// not to the deformer.
    func testRenderUndeformedBaseMesh() async throws {
        await BodyBaseMeshProvider.prepare(for: .male)
        await BodyBaseMeshProvider.prepare(for: .female)

        var columns: [(String, [UIImage])] = []
        for gender in BodyGender.allCases {
            let prepared = try XCTUnwrap(BodyBaseMeshProvider.prepared(for: gender, fatness: 0))
            let positions = prepared.mesh.positions.map { $0 * Float(1.80) }
            let normals = BodyMeshDeformer.normals(for: positions, indices: prepared.mesh.indices)
            let geometry = SCNGeometry(
                sources: [
                    SCNGeometrySource(vertices: positions.map { SCNVector3($0.x, $0.y, $0.z) }),
                    SCNGeometrySource(normals: normals.map { SCNVector3($0.x, $0.y, $0.z) })
                ],
                elements: [SCNGeometryElement(indices: prepared.mesh.indices, primitiveType: .triangles)]
            )
            geometry.subdivisionLevel = 1

            // How many vertices share a position with another? A split seam is
            // exactly what stops `normals(for:indices:)` from averaging across it.
            var buckets: [SIMD3<Int32>: [Int]] = [:]
            for (index, p) in prepared.mesh.positions.enumerated() {
                let key = SIMD3<Int32>(Int32((p.x * 100_000).rounded()),
                                       Int32((p.y * 100_000).rounded()),
                                       Int32((p.z * 100_000).rounded()))
                buckets[key, default: []].append(index)
            }
            let split = buckets.values.filter { $0.count > 1 }
            print("\(gender): \(prepared.mesh.positions.count) wierzcholkow, "
                  + "\(split.count) pozycji wspoldzielonych, "
                  + "\(split.reduce(0) { $0 + $1.count }) wierzcholkow w szwach")
            let clavicle = split.filter {
                let y = prepared.mesh.positions[$0[0]].y
                return y > 0.78 && y < 0.88
            }
            print("  w pasie obojczyka (y 0.78...0.88): \(clavicle.count) pozycji")

            // Welded normals: accumulate across coincident positions, then share.
            let welded = weldedNormals(positions: positions, indices: prepared.mesh.indices)

            for (label, useWelded) in [("obecne normalne", false), ("normalne zespawane", true)] {
                let copy = try XCTUnwrap(geometry.copy() as? SCNGeometry)
                let n = useWelded ? welded : normals
                let rebuilt = SCNGeometry(
                    sources: [
                        SCNGeometrySource(vertices: positions.map { SCNVector3($0.x, $0.y, $0.z) }),
                        SCNGeometrySource(normals: n.map { SCNVector3($0.x, $0.y, $0.z) })
                    ],
                    elements: [SCNGeometryElement(indices: prepared.mesh.indices, primitiveType: .triangles)]
                )
                rebuilt.subdivisionLevel = 1
                let material = SCNMaterial()
                material.lightingModel = .physicallyBased
                material.diffuse.contents = UIColor(red: 0.78, green: 0.76, blue: 0.73, alpha: 1)
                material.roughness.contents = 0.65
                material.metalness.contents = 0.0
                material.isDoubleSided = false
                rebuilt.materials = [material]
                _ = copy
                columns.append(("\(gender) — \(label)", renderRaw(geometry: rebuilt, yaws: [0, .pi / 2])))
            }
        }

        let sheet = contactSheet(title: "Siatka bazowa, zero deformacji", columns: columns)
        let url = outputDirectory().appendingPathComponent("base-mesh.png")
        try XCTUnwrap(sheet.pngData()).write(to: url)
        print("📸 WROTE \(url.path)")
    }

    /// Same scene MannequinView builds, but driven by a geometry handed in.
    private func renderRaw(geometry: SCNGeometry, yaws: [Double],
                           key: Double = 700, fill: Double = 260,
                           rim: Double = 180, ambientIntensity: Double = 300) -> [UIImage] {
        let view = SCNView(frame: CGRect(origin: .zero, size: Self.cell))
        view.scene = SCNScene()
        view.antialiasingMode = .multisampling4X
        view.backgroundColor = UIColor(white: 0.97, alpha: 1)
        view.scene?.background.contents = UIColor(white: 0.97, alpha: 1)

        let node = SCNNode(geometry: geometry)
        view.scene?.rootNode.addChildNode(node)

        let camera = SCNCamera()
        camera.usesOrthographicProjection = true
        camera.orthographicScale = 1.15
        camera.screenSpaceAmbientOcclusionIntensity = 1.4
        camera.screenSpaceAmbientOcclusionRadius = 0.12
        camera.screenSpaceAmbientOcclusionBias = 0.03
        let cameraNode = SCNNode()
        cameraNode.camera = camera
        cameraNode.position = SCNVector3(0, 0.9, 3)
        view.scene?.rootNode.addChildNode(cameraNode)

        for setup in [(key, SCNVector3(2, 3, 3)), (fill, SCNVector3(-3, 2, 1)), (rim, SCNVector3(0, 2, -4))] {
            let light = SCNNode()
            light.light = SCNLight()
            light.light?.type = .directional
            light.light?.intensity = setup.0
            light.position = setup.1
            light.look(at: SCNVector3(0, 0.9, 0))
            view.scene?.rootNode.addChildNode(light)
        }
        let ambient = SCNNode()
        ambient.light = SCNLight()
        ambient.light?.type = .ambient
        ambient.light?.intensity = CGFloat(ambientIntensity)
        view.scene?.rootNode.addChildNode(ambient)

        return yaws.map { yaw in
            node.eulerAngles.y = Float(yaw)
            return view.snapshot()
        }
    }

    /// Same accumulation as `BodyMeshDeformer.normals`, but coincident
    /// positions pool their faces and end up sharing one normal.
    private func weldedNormals(positions: [SIMD3<Float>], indices: [Int32]) -> [SIMD3<Float>] {
        var canonical = [Int](repeating: 0, count: positions.count)
        var seen: [SIMD3<Int32>: Int] = [:]
        for (index, p) in positions.enumerated() {
            let key = SIMD3<Int32>(Int32((p.x * 100_000).rounded()),
                                   Int32((p.y * 100_000).rounded()),
                                   Int32((p.z * 100_000).rounded()))
            if let first = seen[key] { canonical[index] = first } else {
                seen[key] = index
                canonical[index] = index
            }
        }
        var accumulated = [SIMD3<Float>](repeating: .zero, count: positions.count)
        var triangle = 0
        while triangle + 2 < indices.count {
            let i = Int(indices[triangle]), j = Int(indices[triangle + 1]), k = Int(indices[triangle + 2])
            let face = simd_cross(positions[j] - positions[i], positions[k] - positions[i])
            for v in [i, j, k] { accumulated[canonical[v]] += face }
            triangle += 3
        }
        return positions.indices.map {
            let n = accumulated[canonical[$0]]
            return simd_length(n) > 1e-12 ? simd_normalize(n) : SIMD3<Float>(0, 1, 0)
        }
    }
